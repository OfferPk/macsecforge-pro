/\*\*  
 \* MACsecForge Pro \- Multi-Vendor Configuration Synthesizer  
 \* Generates verified production syntax for Cisco IOS-XR, Arista EOS,  
 \* Juniper Junos, Linux Native MACsec, and Prometheus Alerting Rules.  
 \*/

(function (root, factory) {  
  if (typeof module \=== 'object' && module.exports) {  
    module.exports \= factory();  
  } else {  
    root.MACsecConfigSynthesizer \= factory();  
  }  
})(typeof self \!== 'undefined' ? self : this, function () {

  /\*\*  
   \* Generates Cisco IOS-XR MACsec & MKA Configuration  
   \*/  
  function generateCiscoIOSXR(cfg) {  
    const lines \= \[\];  
    lines.push('\! \=======================================================');  
    lines.push('\! Cisco IOS-XR IEEE 802.1AE MACsec / MKA Hardened Configuration');  
    lines.push(\`\! Generated by MACsecForge Pro for Node: \${cfg.hostname || 'CORE-DCI-01'}\`);  
    lines.push('\! \=======================================================');  
    lines.push('');

    const iface \= cfg.interface || 'HundredGigE0/0/0/0';  
    const ckn \= cfg.ckn || '0123456789ABCDEF0123456789ABCDEF0123456789ABCDEF0123456789ABCDEF';  
    const cak \= cfg.cak || 'A1B2C3D4E5F60718293A4B5C6D7E8F90A1B2C3D4E5F60718293A4B5C6D7E8F90';  
    const cipher \= cfg.cipherSuite \=== 'GCM-AES-XPN-256' ? 'gcm-aes-xpn-256' :  
                   (cfg.cipherSuite \=== 'GCM-AES-256' ? 'gcm-aes-256' : 'gcm-aes-128');  
    const replayWin \= cfg.replayProtect ? (cfg.replayWindow || 1024\) : 0;  
    const offset \= parseInt(cfg.confidentialityOffset, 10\) || 0;

    // Key Chain  
    lines.push('key chain MACSEC-KEYCHAIN macsec');  
    lines.push(' key 01');  
    lines.push(\`  key-string password 7 \${cak}\`);  
    lines.push(\`  cryptographic-algorithm \${cipher.includes('256') ? 'aes-256-cmac' : 'aes-128-cmac'}\`);  
    lines.push(' \!');  
    lines.push('\!');  
    lines.push('');

    // MKA Policy  
    lines.push('mka');  
    lines.push(' policy MKA-SECURE-POLICY');  
    lines.push('  key-server priority 16');  
    lines.push(\`  cipher-suite \${cipher}\`);  
    if (cfg.confidentialityOffset \> 0\) {  
      lines.push(\`  confidentiality-offset \${offset}\`);  
    }  
    if (cfg.includeSci) {  
      lines.push('  include-icv-indicator');  
    }  
    lines.push('  macsec-service direct');  
    lines.push(' \!');  
    lines.push('\!');  
    lines.push('');

    // MACsec Policy  
    lines.push('macsec-policy MACSEC-CORE-POLICY');  
    lines.push(\` cipher-suite \${cipher}\`);  
    if (cfg.replayProtect) {  
      lines.push(\` window-size \${replayWin}\`);  
    } else {  
      lines.push(' replay-protection disable');  
    }  
    lines.push('\!');  
    lines.push('');

    // Interface Binding  
    lines.push(\`interface \${iface}\`);  
    lines.push(' description Point-to-Point WAN / DCI Interconnect \[MACsecForge Hardened\]');  
    lines.push(\` mtu \${cfg.interfaceMtu || 1532}\`);  
    lines.push(' macsec');  
    lines.push('  policy MACSEC-CORE-POLICY');  
    lines.push(' \!');  
    lines.push(' mka');  
    lines.push('  policy MKA-SECURE-POLICY');  
    lines.push('  pre-shared-key');  
    lines.push(\`   ckn \${ckn}\`);  
    lines.push('   key-chain MACSEC-KEYCHAIN');  
    lines.push('  \!');  
    lines.push(' \!');  
    lines.push('\!');

    return lines.join('\\n');  
  }

  /\*\*  
   \* Generates Arista EOS MACsec Configuration  
   \*/  
  function generateAristaEOS(cfg) {  
    const lines \= \[\];  
    lines.push('\! \=======================================================');  
    lines.push('\! Arista EOS 802.1AE MACsec Profile & Interface Configuration');  
    lines.push(\`\! Generated by MACsecForge Pro for Node: \${cfg.hostname || 'LEAF-DCI-01'}\`);  
    lines.push('\! \=======================================================');  
    lines.push('');

    const iface \= cfg.interface || 'Ethernet1';  
    const ckn \= cfg.ckn || '0123456789ABCDEF0123456789ABCDEF0123456789ABCDEF0123456789ABCDEF';  
    const cak \= cfg.cak || 'A1B2C3D4E5F60718293A4B5C6D7E8F90A1B2C3D4E5F60718293A4B5C6D7E8F90';  
    const cipher \= cfg.cipherSuite \=== 'GCM-AES-XPN-256' ? 'aes256-gcm-xpn' :  
                   (cfg.cipherSuite \=== 'GCM-AES-256' ? 'aes256-gcm' : 'aes128-gcm');  
    const replayWin \= cfg.replayProtect ? (cfg.replayWindow || 1024\) : 0;

    // MACsec Profile  
    lines.push('macsec');  
    lines.push(' profile MACSEC-PRO');  
    lines.push(\`  cipher \${cipher}\`);  
    lines.push('  key-server priority 16');  
    lines.push(\`  mka key ckn \${ckn} cak 7 \${cak}\`);  
    if (cfg.replayProtect) {  
      lines.push(\`  replay-protection window-size \${replayWin}\`);  
    } else {  
      lines.push('  no replay-protection');  
    }  
    if (cfg.failClosed) {  
      lines.push('  traffic-unprotected drop');  
    } else {  
      lines.push('  traffic-unprotected allow');  
    }  
    lines.push(' \!');  
    lines.push('\!');  
    lines.push('');

    // Interface  
    lines.push(\`interface \${iface}\`);  
    lines.push(' description Backbone-DCI-Trunk \[MACsecForge\]');  
    lines.push(\` mtu \${cfg.interfaceMtu || 1532}\`);  
    lines.push(' macsec profile MACSEC-PRO');  
    lines.push('\!');

    return lines.join('\\n');  
  }

  /\*\*  
   \* Generates Juniper Junos MACsec Configuration  
   \*/  
  function generateJuniperJunos(cfg) {  
    const lines \= \[\];  
    lines.push('\# \=======================================================');  
    lines.push('\# Juniper Junos Security MACsec Configuration');  
    lines.push(\`\# Generated by MACsecForge Pro for Node: \${cfg.hostname || 'JUNOS-WAN-01'}\`);  
    lines.push('\# \=======================================================');  
    lines.push('');

    const iface \= cfg.interface || 'et-0/0/0';  
    const ckn \= cfg.ckn || '0123456789ABCDEF0123456789ABCDEF0123456789ABCDEF0123456789ABCDEF';  
    const cak \= cfg.cak || 'A1B2C3D4E5F60718293A4B5C6D7E8F90A1B2C3D4E5F60718293A4B5C6D7E8F90';  
    const cipher \= cfg.cipherSuite \=== 'GCM-AES-XPN-256' ? 'gcm-aes-xpn-256' :  
                   (cfg.cipherSuite \=== 'GCM-AES-256' ? 'gcm-aes-256' : 'gcm-aes-128');  
    const replayWin \= cfg.replayProtect ? (cfg.replayWindow || 1024\) : 0;

    lines.push(\`set security macsec connectivity-association CA-DCI-01 security-mode static-cak\`);  
    lines.push(\`set security macsec connectivity-association CA-DCI-01 cipher-suite \${cipher}\`);  
    lines.push(\`set security macsec connectivity-association CA-DCI-01 pre-shared-key ckn \${ckn}\`);  
    lines.push(\`set security macsec connectivity-association CA-DCI-01 pre-shared-key cak "\${cak}"\`);  
    if (cfg.replayProtect) {  
      lines.push(\`set security macsec connectivity-association CA-DCI-01 replay-protect window-size \${replayWin}\`);  
    }  
    if (cfg.failClosed) {  
      lines.push(\`set security macsec connectivity-association CA-DCI-01 exclude-traffic-unencrypted\`);  
    }  
    lines.push(\`set security macsec interfaces \${iface} connectivity-association CA-DCI-01\`);  
    lines.push(\`set interfaces \${iface} mtu \${cfg.interfaceMtu || 1532}\`);

    return lines.join('\\n');  
  }

  /\*\*  
   \* Generates Linux Native MACsec (\`ip link\` / \`wpa\_supplicant\`) Configuration  
   \*/  
  function generateLinuxMACsec(cfg) {  
    const lines \= \[\];  
    lines.push('\# \=======================================================');  
    lines.push('\# Linux Native Kernel MACsec (iproute2) Configuration');  
    lines.push(\`\# Generated by MACsecForge Pro for Host: \${cfg.hostname || 'linux-edge-01'}\`);  
    lines.push('\# \=======================================================');  
    lines.push('');

    const iface \= cfg.interface || 'eth0';  
    const ckn \= (cfg.ckn || '0123456789ABCDEF0123456789ABCDEF0123456789ABCDEF0123456789ABCDEF').toLowerCase();  
    const cak \= (cfg.cak || 'A1B2C3D4E5F60718293A4B5C6D7E8F90A1B2C3D4E5F60718293A4B5C6D7E8F90').toLowerCase();  
    const cipher \= cfg.cipherSuite \=== 'GCM-AES-256' || cfg.cipherSuite \=== 'GCM-AES-XPN-256' ? 'gcm-aes-256' : 'gcm-aes-128';  
    const replayWin \= cfg.replayProtect ? (cfg.replayWindow || 1024\) : 0;

    lines.push('\# 1\. Create MACsec virtual interface over underlying physical link');  
    lines.push(\`sudo ip link add link \${iface} macsec0 type macsec \\\\\`);  
    lines.push(\`  cipher \${cipher} \\\\\`);  
    lines.push(\`  encrypt on \\\\\`);  
    lines.push(\`  send\_sci \${cfg.includeSci ? 'on' : 'off'} \\\\\`);  
    lines.push(\`  replay \${cfg.replayProtect ? 'on' : 'off'} window \${replayWin}\`);  
    lines.push('');  
    lines.push('\# 2\. Configure Transmit Secure Association (TX SA 0)');  
    lines.push(\`sudo ip macsec add macsec0 tx sa 0 pn 1 on key 01 \${cak.substr(0, 64)}\`);  
    lines.push('');  
    lines.push('\# 3\. Configure Receive Secure Channel (RX SC) and SA 0');  
    const peerSci \= cfg.peerSci || '00:11:22:33:44:56:00:01';  
    lines.push(\`sudo ip macsec add macsec0 rx port 1 address \${peerSci.split(':').slice(0, 6).join(':')}\`);  
    lines.push(\`sudo ip macsec add macsec0 rx port 1 address \${peerSci.split(':').slice(0, 6).join(':')} sa 0 pn 1 on key 01 \${cak.substr(0, 64)}\`);  
    lines.push('');  
    lines.push('\# 4\. Bring up MACsec link and set MTU');  
    lines.push(\`sudo ip link set dev \${iface} mtu \${cfg.interfaceMtu || 1532}\`);  
    lines.push('sudo ip link set dev macsec0 up');  
    lines.push(\`sudo ip addr add \${cfg.localIp || '10.50.0.1'}/30 dev macsec0\`);

    return lines.join('\\n');  
  }

  /\*\*  
   \* Generates Prometheus & Alertmanager MACsec Metrics Exporter Rules  
   \*/  
  function generatePrometheusRules(cfg) {  
    const lines \= \[\];  
    lines.push('\# \=======================================================');  
    lines.push('\# Prometheus & OpenTelemetry MACsec Alerting Rules');  
    lines.push('\# Standards: IEEE 802.1AE-2006 / RFC 7634');  
    lines.push('\# \=======================================================');  
    lines.push('');  
    lines.push('groups:');  
    lines.push('  \- name: macsec\_zero\_trust\_alerts');  
    lines.push('    rules:');  
    lines.push('      \- alert: MACsecSessionDown');  
    lines.push('        expr: macsec\_mka\_session\_state \!= 3');  
    lines.push('        for: 10s');  
    lines.push('        labels:');  
    lines.push('          severity: critical');  
    lines.push('        annotations:');  
    lines.push(\`          summary: "MACsec session down on {{ \$labels.instance }} ({{ \$labels.interface }})"\`);  
    lines.push('          description: "MKA session dropped. Fail-closed policy is dropping transit frames to prevent unencrypted cleartext leakage."');  
    lines.push('');  
    lines.push('      \- alert: MACsecICVErrorSpike');  
    lines.push('        expr: rate(macsec\_in\_pkts\_bad\_tag\_total\[1m\]) \> 5');  
    lines.push('        for: 30s');  
    lines.push('        labels:');  
    lines.push('          severity: critical');  
    lines.push('        annotations:');  
    lines.push('          summary: "MACsec Integrity Check Value (ICV) failure rate high"');  
    lines.push('          description: "Potential optical fiber tapping, physical MITM tampering, or cipher key desynchronization detected."');  
    lines.push('');  
    lines.push('      \- alert: MACsecReplayDropSpike');  
    lines.push('        expr: rate(macsec\_in\_pkts\_late\_total\[1m\]) \> 20');  
    lines.push('        for: 1m');  
    lines.push('        labels:');  
    lines.push('          severity: warning');  
    lines.push('        annotations:');  
    lines.push('          summary: "MACsec late/replay packet drops detected"');  
    lines.push('          description: "Packets arriving outside the configured replay window. Review QoS queue priority or expand replay window size."');  
    lines.push('');  
    lines.push('      \- alert: MACsecPNExhaustionThreshold');  
    lines.push('        expr: macsec\_tx\_packet\_number \> 3221225471  \# 75% of 2^32');  
    lines.push('        for: 10s');  
    lines.push('        labels:');  
    lines.push('          severity: warning');  
    lines.push('        annotations:');  
    lines.push('          summary: "MACsec Packet Number reaching rollover limit"');  
    lines.push('          description: "32-bit PN has exceeded 75% of maximum capacity. Verify SAK rekeying or transition to GCM-AES-XPN-256."');

    return lines.join('\\n');  
  }

  return {  
    generateCiscoIOSXR: generateCiscoIOSXR,  
    generateAristaEOS: generateAristaEOS,  
    generateJuniperJunos: generateJuniperJunos,  
    generateLinuxMACsec: generateLinuxMACsec,  
    generatePrometheusRules: generatePrometheusRules  
  };  
});
